version: '1'
name: streams-features-identification
description: Identify knowledge indicator features for a stream using LLM and computed analysis
enabled: true
triggers:
  - type: manual
inputs:
  - name: stream_name
    type: string
    required: true
  - name: start
    type: number
    required: true
  - name: end
    type: number
    required: true

settings:
  concurrency:
    key: "streams-features-${{ inputs.stream_name }}"
    max: 1
    strategy: cancel-in-progress
  timeout: 30m

steps:
  - name: fetch_samples
    type: kibana.request
    with:
      method: POST
      path: /internal/streams/{{ inputs.stream_name }}/features/_workflow/fetch_samples
      headers:
        x-elastic-internal-origin: kibana
      body:
        start: ${{ inputs.start }}
        end: ${{ inputs.end }}

  - name: identify_features
    type: kibana.request
    with:
      method: POST
      path: /internal/streams/{{ inputs.stream_name }}/features/_workflow/identify
      headers:
        x-elastic-internal-origin: kibana
      body:
        documents: ${{ steps.fetch_samples.output.documents }}

  - name: computed_features
    type: kibana.request
    with:
      method: POST
      path: /internal/streams/{{ inputs.stream_name }}/features/_workflow/computed
      headers:
        x-elastic-internal-origin: kibana
      body:
        start: ${{ inputs.start }}
        end: ${{ inputs.end }}

  - name: persist_features
    type: kibana.request
    with:
      method: POST
      path: /internal/streams/{{ inputs.stream_name }}/features/_workflow/persist
      headers:
        x-elastic-internal-origin: kibana
      body:
        inferred_features: ${{ steps.identify_features.output.features }}
        computed_features: ${{ steps.computed_features.output.features }}
        ignored_features: ${{ steps.identify_features.output.ignored_features }}
